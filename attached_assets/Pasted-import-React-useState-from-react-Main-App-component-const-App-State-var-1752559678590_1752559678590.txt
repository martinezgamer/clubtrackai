import React, { useState } from 'react';

// Main App component
const App = () => {
    // State variables for managing user input, generated content, and loading states
    const [postIdeaPrompt, setPostIdeaPrompt] = useState(''); // Text input for post ideas
    const [imageGenerationPrompt, setImageGenerationPrompt] = useState(''); // Text input for image generation
    const [postIdeas, setPostIdeas] = useState([]); // Array to store generated text post ideas
    const [generatedImageUrl, setGeneratedImageUrl] = useState(''); // URL of the generated image
    const [isLoadingIdeas, setIsLoadingIdeas] = useState(false); // Loading state for text generation
    const [isLoadingImage, setIsLoadingImage] = useState(false); // Loading state for image generation
    const [errorMessage, setErrorMessage] = useState(''); // To display any errors
    const [uploadedImagePreview, setUploadedImagePreview] = useState(''); // URL for uploaded image preview (Data URL)
    const [copiedIndex, setCopiedIndex] = useState(null); // To show "Copied!" message for a specific idea

    // Function to handle image file upload
    const handleImageUpload = (event) => {
        const file = event.target.files[0];
        if (file) {
            // Check if the file is an image
            if (!file.type.startsWith('image/')) {
                setErrorMessage('Please upload an image file (e.g., JPG, PNG, GIF).');
                setUploadedImagePreview('');
                return;
            }
            setErrorMessage(''); // Clear any previous error
            const reader = new FileReader();
            reader.onloadend = () => {
                setUploadedImagePreview(reader.result); // Store the Data URL
            };
            reader.readAsDataURL(file);
        } else {
            setUploadedImagePreview('');
        }
    };

    // Function to copy text to clipboard
    const copyToClipboard = (text, index) => {
        try {
            const textarea = document.createElement('textarea');
            textarea.value = text;
            document.body.appendChild(textarea);
            textarea.select();
            document.execCommand('copy'); // Use document.execCommand for broader compatibility in iframes
            document.body.removeChild(textarea);

            setCopiedIndex(index); // Set index to show "Copied!" message
            setTimeout(() => setCopiedIndex(null), 2000); // Hide after 2 seconds
        } catch (err) {
            console.error('Failed to copy text: ', err);
            setErrorMessage('Failed to copy text. Please copy manually.');
        }
    };

    // Function to handle generating post ideas using the Gemini API
    const generatePostIdeas = async () => {
        setIsLoadingIdeas(true);
        setErrorMessage('');
        setPostIdeas([]); // Clear previous ideas

        const chatHistory = [];
        const userParts = [];

        // If an image is uploaded, prepare it for the API call
        if (uploadedImagePreview) {
            const mimeTypeMatch = uploadedImagePreview.match(/^data:(image\/(jpeg|png|gif));base64,/);
            if (!mimeTypeMatch) {
                setErrorMessage('Uploaded file is not a valid image format (JPG, PNG, GIF).');
                setIsLoadingIdeas(false);
                return;
            }
            const mimeType = mimeTypeMatch[1]; // e.g., "image/png"
            const base64Data = uploadedImagePreview.split(',')[1]; // Get the base64 part

            userParts.push({
                inlineData: {
                    mimeType: mimeType,
                    data: base64Data
                }
            });

            // Adjust the text prompt to tell the model to consider the image
            userParts.push({
                text: `Considering the uploaded image${postIdeaPrompt ? ` and the following description: "${postIdeaPrompt}"` : ''}, generate 3-5 creative social media post ideas for Facebook and Instagram. Include ideas for captions, relevant hashtags, and a call to action. Format each idea clearly.`
            });
        } else if (postIdeaPrompt.trim()) {
            // If no image is uploaded, just use the text prompt
            userParts.push({ text: `Generate 3-5 creative social media post ideas for Facebook and Instagram based on the following topic/description: "${postIdeaPrompt}". Include ideas for captions, relevant hashtags, and a call to action. Format each idea clearly.` });
        } else {
            setErrorMessage('Please enter a prompt for post ideas or upload an image.');
            setIsLoadingIdeas(false);
            return;
        }

        chatHistory.push({ role: "user", parts: userParts });
        const payload = { contents: chatHistory };
        const apiKey = ""; // API key will be provided by the environment
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;

        try {
            // Make the API call
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) {
                const errorData = await response.json();
                throw new Error(errorData.error?.message || 'Failed to generate post ideas.');
            }

            const result = await response.json();

            // Process the API response
            if (result.candidates && result.candidates.length > 0 &&
                result.candidates[0].content && result.candidates[0].content.parts &&
                result.candidates[0].content.parts.length > 0) {
                const text = result.candidates[0].content.parts[0].text;
                // Split the text into individual ideas for better display
                const ideasArray = text.split(/\n\n(?=\d+\.)/).filter(idea => idea.trim() !== '');
                setPostIdeas(ideasArray);
            } else {
                setErrorMessage('No post ideas generated. Please try a different prompt.');
            }
        } catch (error) {
            console.error('Error generating post ideas:', error);
            setErrorMessage(`Error: ${error.message}`);
        } finally {
            setIsLoadingIdeas(false);
        }
    };

    // Function to handle generating an image using the Imagen API
    const generateImage = async () => {
        setIsLoadingImage(true);
        setErrorMessage('');
        setGeneratedImageUrl(''); // Clear previous image

        if (!imageGenerationPrompt.trim()) {
            setErrorMessage('Please enter a prompt for image generation.');
            setIsLoadingImage(false);
            return;
        }

        try {
            // Prepare the payload for the Imagen API call
            const payload = { instances: { prompt: imageGenerationPrompt }, parameters: { "sampleCount": 1 } };
            const apiKey = ""; // API key will be provided by the environment
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/imagen-3.0-generate-002:predict?key=${apiKey}`;

            // Make the API call
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) {
                const errorData = await response.json();
                throw new Error(errorData.error?.message || 'Failed to generate image.');
            }

            const result = await response.json();

            // Process the API response
            if (result.predictions && result.predictions.length > 0 && result.predictions[0].bytesBase64Encoded) {
                const imageUrl = `data:image/png;base64,${result.predictions[0].bytesBase64Encoded}`;
                setGeneratedImageUrl(imageUrl);
            } else {
                setErrorMessage('No image generated. Please try a different prompt.');
            }
        } catch (error) {
            console.error('Error generating image:', error);
            setErrorMessage(`Error: ${error.message}`);
        } finally {
            setIsLoadingImage(false);
        }
    };

    return (
        <div className="min-h-screen bg-gradient-to-br from-purple-900 via-fuchsia-800 to-pink-600 p-4 font-sans text-gray-800 flex items-center justify-center">
            <div className="bg-white p-8 rounded-xl shadow-2xl w-full max-w-4xl space-y-8">
                <h1 className="text-4xl font-extrabold text-center text-fuchsia-500 mb-8 drop-shadow-lg">
                    Social Media Content Creator
                </h1>

                {/* Error Message Display */}
                {errorMessage && (
                    <div className="bg-red-100 border border-red-400 text-red-700 px-4 py-3 rounded-md relative mb-8" role="alert">
                        <strong className="font-bold">Error!</strong>
                        <span className="block sm:inline ml-2">{errorMessage}</span>
                    </div>
                )}

                {/* Image Upload Section */}
                <div className="space-y-4 p-6 bg-gray-100 rounded-lg shadow-inner mb-8 border border-gray-200">
                    <h2 className="text-2xl font-bold text-gray-700">Upload Image/Flyer (for AI Analysis & Reference)</h2>
                    <input
                        type="file"
                        accept="image/*"
                        onChange={handleImageUpload}
                        className="w-full text-gray-700 file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-fuchsia-200 file:text-fuchsia-800 hover:file:bg-fuchsia-300 transition duration-200 ease-in-out"
                    />
                    {uploadedImagePreview && (
                        <div className="mt-4 text-center p-4 bg-white rounded-lg shadow-md border border-gray-200">
                            <h3 className="text-xl font-semibold text-gray-600 mb-2">Uploaded Image Preview:</h3>
                            <img
                                src={uploadedImagePreview}
                                alt="Uploaded Preview"
                                className="max-w-full h-auto rounded-lg shadow-md mx-auto border border-gray-300"
                                style={{ maxHeight: '300px' }} // Limit height for preview
                            />
                            <p className="text-sm text-gray-500 mt-2">
                                *The AI will analyze this image when generating post ideas.*
                            </p>
                        </div>
                    )}
                </div>

                {/* Post Idea Generation Section */}
                <div className="space-y-4 p-6 bg-fuchsia-50 rounded-lg shadow-inner mb-8 border border-fuchsia-100">
                    <h2 className="text-2xl font-bold text-fuchsia-700">Generate Post Ideas</h2>
                    <textarea
                        className="w-full p-3 border border-fuchsia-300 rounded-lg focus:ring-2 focus:ring-fuchsia-500 focus:border-transparent transition duration-200 ease-in-out resize-y min-h-[100px]"
                        placeholder="Optional: Add more details about the post idea you want, e.g., 'Focus on luxury and exclusivity' or 'Promote our weekend special'."
                        value={postIdeaPrompt}
                        onChange={(e) => setPostIdeaPrompt(e.target.value)}
                        rows="4"
                    ></textarea>
                    <button
                        onClick={generatePostIdeas}
                        className="w-full bg-fuchsia-600 text-white py-3 px-6 rounded-lg font-semibold text-lg hover:bg-fuchsia-700 transition duration-300 ease-in-out shadow-lg hover:shadow-xl focus:outline-none focus:ring-2 focus:ring-fuchsia-500 focus:ring-opacity-75 transform hover:-translate-y-0.5"
                        disabled={isLoadingIdeas}
                    >
                        {isLoadingIdeas ? (
                            <div className="flex items-center justify-center">
                                <svg className="animate-spin -ml-1 mr-3 h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
                                    <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
                                    <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
                                </svg>
                                Generating Ideas...
                            </div>
                        ) : 'Generate Post Ideas'}
                    </button>
                </div>

                {/* Display Generated Post Ideas */}
                {postIdeas.length > 0 && (
                    <div className="space-y-4 p-6 bg-white rounded-lg shadow-lg border border-fuchsia-200 mb-8">
                        <h2 className="text-2xl font-bold text-fuchsia-700">Generated Post Ideas</h2>
                        <div className="space-y-4">
                            {postIdeas.map((idea, index) => (
                                <div key={index} className="bg-fuchsia-50 p-4 rounded-md border border-fuchsia-100 shadow-sm flex flex-col sm:flex-row justify-between items-start sm:items-center">
                                    <p className="text-gray-700 whitespace-pre-wrap flex-grow mb-2 sm:mb-0">{idea}</p>
                                    <div className="flex items-center">
                                        <button
                                            onClick={() => copyToClipboard(idea, index)}
                                            className="ml-0 sm:ml-4 bg-fuchsia-500 text-white py-1 px-3 rounded-md text-sm font-semibold hover:bg-fuchsia-600 transition duration-200 ease-in-out focus:outline-none focus:ring-2 focus:ring-fuchsia-400 focus:ring-opacity-75"
                                        >
                                            {copiedIndex === index ? 'Copied!' : 'Copy'}
                                        </button>
                                    </div>
                                </div>
                            ))}
                        </div>
                    </div>
                )}

                {/* Image Generation Section */}
                <div className="space-y-4 p-6 bg-indigo-50 rounded-lg shadow-inner border border-indigo-100">
                    <h2 className="text-2xl font-bold text-indigo-700">Generate Image for Post</h2>
                    <textarea
                        className="w-full p-3 border border-indigo-300 rounded-lg focus:ring-2 focus:ring-indigo-500 focus:border-transparent transition duration-200 ease-in-out resize-y min-h-[100px]"
                        placeholder="e.g., 'A vibrant summer beach scene with people playing volleyball, realistic, high detail', 'Abstract art with swirling blue and gold colors', 'Describe the image you want based on your uploaded flyer...'"
                        value={imageGenerationPrompt}
                        onChange={(e) => setImageGenerationPrompt(e.target.value)}
                        rows="4"
                    ></textarea>
                    <button
                        onClick={generateImage}
                        className="w-full bg-indigo-600 text-white py-3 px-6 rounded-lg font-semibold text-lg hover:bg-indigo-700 transition duration-300 ease-in-out shadow-lg hover:shadow-xl focus:outline-none focus:ring-2 focus:ring-indigo-500 focus:ring-opacity-75 transform hover:-translate-y-0.5"
                        disabled={isLoadingImage}
                    >
                        {isLoadingImage ? (
                            <div className="flex items-center justify-center">
                                <svg className="animate-spin -ml-1 mr-3 h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
                                    <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
                                    <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
                                </svg>
                                Generating Image...
                            </div>
                        ) : 'Generate Image'}
                    </button>
                </div>

                {/* Display Generated Image */}
                {generatedImageUrl && (
                    <div className="p-6 bg-white rounded-lg shadow-lg border border-indigo-200">
                        <h2 className="text-2xl font-bold text-indigo-700 mb-4">Generated Image</h2>
                        <img
                            src={generatedImageUrl}
                            alt="Generated Post Visual"
                            className="max-w-full h-auto rounded-lg shadow-md mx-auto border border-gray-300"
                            onError={(e) => { e.target.onerror = null; e.target.src = "https://placehold.co/600x400/E0E7FF/4F46E5?text=Image+Load+Error"; setErrorMessage('Failed to load generated image.'); }}
                        />
                    </div>
                )}
            </div>
        </div>
    );
};

export default App;
